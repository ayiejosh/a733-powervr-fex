# glbench GL results, 2026-09-23 - four GL paths on one board
#
# Same binary (bench/glbench.c, 1280x720, fullscreen triangle, ALU-loop fragment
# shader) for every row, and the same surfaceless EGL config, because that is the
# only config that exists on all four paths: with the GBM+pbuffer config, zink
# returns EGL_BAD_CONFIG ("no context 0x3005") and the run produces no result at
# all rather than failing loudly. Frame counts fall with the shader loop.
#
#   stock (no zink):  env -i PATH=/usr/bin:/bin HOME=/home/radxa \
#                       MESA_LOADER_DRIVER_OVERRIDE=llvmpipe /tmp/glbench none <L> <F>
#   vendor native:    env -i PATH=/usr/bin:/bin HOME=/home/radxa /tmp/glbench none <L> <F>
#                     (/usr/local/lib is in ld.so.conf, so the vendor EGL wins by default)
#   repo zink:        glrun /tmp/glbench none <L> <F>
#                     (zink over the vendor ICD + the feature-strip layer)
#   local zink:       LD_LIBRARY_PATH=/home/radxa/mesa/inst-gl/usr/local/lib/aarch64-linux-gnu \
#                     LIBGL_DRIVERS_PATH=/home/radxa/mesa/gldri MESA_LOADER_DRIVER_OVERRIDE=zink \
#                     GALLIUM_DRIVER=zink MESA_GLES_VERSION_OVERRIDE=3.2 \
#                     VK_ICD_FILENAMES=<build-x11 icd> /tmp/glbench none <L> <F>
#
# clock: 1104 MHz for every row except the two marked 552 MHz (the no-overlay
# default, set live through bench/clkctl).
